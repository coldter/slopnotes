**Problem:** Auto-assign quality inspections, customer complaints, and franchise audit visits to the right people — no manager hand-picking an owner per record.

**Solution:** Frappe's **Assignment Rule** DocType. Each rule watches a target DocType, evaluates an `assign_condition` on every save, and assigns matching documents using one of four strategies. All four patterns live in one setup script, run once with `bench execute scoopjoy.scoopjoy.setup.assignment_rules.create_assignment_rules`:

```python title="scoopjoy/scoopjoy/setup/assignment_rules.py"
import frappe


def create_assignment_rules():
    """Create assignment rules for ScoopJoy operations.
    Run: bench execute scoopjoy.scoopjoy.setup.assignment_rules.create_assignment_rules
    """
    _create_qa_round_robin()
    _create_complaint_load_balance()
    _create_audit_field_based()
    _create_complex_condition_rule()
    frappe.db.commit()
```

### Pattern 1: Round Robin — QA inspections

Distributes evenly: Frappe tracks `last_user` on the rule and cycles through `users`. `assignment_days` restricts assignment to QA working days:

```python title="scoopjoy/scoopjoy/setup/assignment_rules.py" {8,13}
def _create_qa_round_robin():
    """Round Robin: distribute QA inspections evenly across the QA team."""
    if frappe.db.exists("Assignment Rule", "QA Inspection Round Robin"):
        return

    frappe.get_doc({
        "doctype": "Assignment Rule",
        "name": "QA Inspection Round Robin",
        "document_type": "Quality Inspection",
        "assign_condition": "doc.docstatus == 0 and doc.status == 'Draft'",
        "priority": 0,
        "disabled": 0,

        "rule": "Round Robin",

        "assignment_days": [
            {"day": "Monday"}, {"day": "Tuesday"}, {"day": "Wednesday"},
            {"day": "Thursday"}, {"day": "Friday"}, {"day": "Saturday"},
        ],

        "users": [
            {"user": "qa.inspector1@scoopjoy.com"},
            {"user": "qa.inspector2@scoopjoy.com"},
            {"user": "qa.inspector3@scoopjoy.com"},
        ],

        "due_date_based_on": "Creation Date",
        "due_date_days": 2,

        "close_condition": "doc.docstatus == 1",

        "description": "Assigns QA inspections to inspectors in round-robin order",
    }).insert(ignore_permissions=True)
```

### Pattern 2: Load Balancing — customer complaints

Counts each user's open ToDos and assigns to whoever has the fewest — ideal for support. `unassign_condition` releases the assignment if the issue type changes:

```python title="scoopjoy/scoopjoy/setup/assignment_rules.py" {14,28}
def _create_complaint_load_balance():
    """Load Balancing: assign complaints to the least-busy support agent."""
    if frappe.db.exists("Assignment Rule", "Complaint Load Balancer"):
        return

    frappe.get_doc({
        "doctype": "Assignment Rule",
        "name": "Complaint Load Balancer",
        "document_type": "Issue",
        "assign_condition": (
            "doc.status == 'Open' and "
            "doc.issue_type == 'Customer Complaint'"
        ),
        "priority": 0,
        "disabled": 0,

        "rule": "Load Balancing",  # user with fewest open assignments

        "users": [
            {"user": "support.agent1@scoopjoy.com"},
            {"user": "support.agent2@scoopjoy.com"},
            {"user": "support.agent3@scoopjoy.com"},
            {"user": "support.agent4@scoopjoy.com"},
        ],

        "due_date_based_on": "Creation Date",
        "due_date_days": 1,

        "close_condition": "doc.status in ('Closed', 'Resolved')",
        "unassign_condition": "doc.issue_type != 'Customer Complaint'",

        "description": "Assigns customer complaints to the support agent with fewest open tickets",
    }).insert(ignore_permissions=True)
```

### Pattern 3: Based on Field — franchise audit

Reads a User link field from the document itself, so each outlet's audit goes to its designated auditor. `field` names that link field; `priority: 1` out-ranks the round-robin rule (priority 0) when both match:

```python title="scoopjoy/scoopjoy/setup/assignment_rules.py" {18,19}
def _create_audit_field_based():
    """Based on Field: assign audit to the outlet's designated auditor."""
    if frappe.db.exists("Assignment Rule", "Franchise Audit Auditor"):
        return

    frappe.get_doc({
        "doctype": "Assignment Rule",
        "name": "Franchise Audit Auditor",
        "document_type": "Quality Inspection",
        "assign_condition": (
            "doc.inspection_type == 'Incoming' and "
            "doc.custom_is_franchise_audit == 1"
        ),
        "priority": 1,  # Higher priority than the round-robin rule
        "disabled": 0,

        "rule": "Based on Field",
        "field": "custom_designated_auditor",  # Link field on Quality Inspection

        "due_date_based_on": "Creation Date",
        "due_date_days": 7,

        "close_condition": "doc.docstatus == 1",

        "description": "Assigns franchise audits to the outlet's designated auditor",
    }).insert(ignore_permissions=True)
```

### Pattern 4: Complex condition — item group AND territory

Encode several attributes in `assign_condition` itself — ice-cream inspections in the South zone go to the South QA team, etc. These rules use the highest `priority` (2) because they're the most specific:

```python title="scoopjoy/scoopjoy/setup/assignment_rules.py" {8,13}
def _create_complex_condition_rule():
    """Complex condition: item group AND territory.
    Ice cream in South zone -> south.qa@scoopjoy.com, etc.
    """
    rules = [
        {
            "name": "QA South Zone Ice Cream",
            "condition": (
                "doc.item_group == 'Ice Cream' and "
                "frappe.db.get_value('Warehouse', doc.warehouse, 'custom_zone') == 'South'"
            ),
            "users": [{"user": "south.qa@scoopjoy.com"}],
        },
        {
            "name": "QA North Zone Ice Cream",
            "condition": (
                "doc.item_group == 'Ice Cream' and "
                "frappe.db.get_value('Warehouse', doc.warehouse, 'custom_zone') == 'North'"
            ),
            "users": [{"user": "north.qa@scoopjoy.com"}],
        },
        {
            "name": "QA South Zone Toppings",
            "condition": (
                "doc.item_group == 'Toppings' and "
                "frappe.db.get_value('Warehouse', doc.warehouse, 'custom_zone') == 'South'"
            ),
            "users": [{"user": "south.qa.toppings@scoopjoy.com"}],
        },
        {
            "name": "QA North Zone Toppings",
            "condition": (
                "doc.item_group == 'Toppings' and "
                "frappe.db.get_value('Warehouse', doc.warehouse, 'custom_zone') == 'North'"
            ),
            "users": [{"user": "north.qa.toppings@scoopjoy.com"}],
        },
    ]

    for rule_def in rules:
        if frappe.db.exists("Assignment Rule", rule_def["name"]):
            continue

        frappe.get_doc({
            "doctype": "Assignment Rule",
            "name": rule_def["name"],
            "document_type": "Quality Inspection",
            "assign_condition": rule_def["condition"],
            "priority": 2,  # Highest priority — most specific
            "disabled": 0,
            "rule": "Round Robin",
            "users": rule_def["users"],
            "due_date_based_on": "Creation Date",
            "due_date_days": 3,
            "close_condition": "doc.docstatus == 1",
            "description": f"Zone/item-specific QA assignment: {rule_def['name']}",
        }).insert(ignore_permissions=True)
```

:::note[Why this works]
Assignment Rules evaluate `assign_condition` on every save of the target DocType; higher `priority` wins. Round Robin cycles through the user list via an internal `last_user`. Load Balancing counts open ToDos per user and picks the fewest. "Based on Field" reads a User link field off the document itself.
:::

:::caution[Common mistakes]
- Overlapping rules without explicit `priority` — when two rules match the same document, the higher-priority one wins; without priorities the behavior is unpredictable.
- Heavy DB queries inside `assign_condition` — it runs as `frappe.safe_eval`, so pre-compute values into custom fields instead.
:::
