Backups and observability are core operational requirements. This covers `bench backup`, S3 offsite rotation, a disaster-recovery runbook, and a Prometheus/Grafana monitoring stack.

## Bench backup: what gets backed up

`bench backup` produces three files:

| File | Contents | Format |
|------|----------|--------|
| `*-database.sql.gz` | Full MariaDB dump | Compressed SQL |
| `*-files.tar` | Public files (uploads, images) | Tar archive |
| `*-private-files.tar` | Private files (attachments, PDFs) | Tar archive |

```bash
bench --site scoopjoy.com backup                              # DB only
bench --site scoopjoy.com backup --with-files                 # recommended
bench backup-all-sites                                         # all sites
bench --site scoopjoy.com backup --with-files --compress
```

Files land in `sites/<site-name>/private/backups/`.

:::danger[Always back up with --with-files]
The database alone is **incomplete** — uploads, images, and private attachments live on the filesystem. A DB-only backup restores to a site with missing files, and you won't notice until a user opens an invoice with a vanished attachment.
:::

## Automated backups

`bench setup production` adds a crontab entry running backups every 6 hours — check with `crontab -l | grep backup`. For a custom schedule:

```bash
crontab -e

# Backup every 6 hours with files
0 */6 * * * cd ~/frappe-bench && ~/frappe-bench/env/bin/bench --site scoopjoy.com backup --with-files >> ~/frappe-bench/logs/backup.log 2>&1

# Backup all sites at 2 AM daily
0 2 * * * cd ~/frappe-bench && ~/frappe-bench/env/bin/bench backup-all-sites --with-files >> ~/frappe-bench/logs/backup-all.log 2>&1
```

## Offsite backup to S3

### Built-in S3 settings

```bash
bench --site scoopjoy.com set-config backup_s3_bucket "scoopjoy-erp-backups"
bench --site scoopjoy.com set-config backup_s3_region "us-east-1"
bench --site scoopjoy.com set-config backup_s3_access_key "AKIA..."
bench --site scoopjoy.com set-config backup_s3_secret_key "your-secret-key"
```

### Custom script with rotation

For full control (multi-site, local + S3 retention), a script that backs up each site, uploads to S3 including `site_config.json`, then prunes past retention windows:

```bash title="~/scripts/backup-to-s3.sh"
#!/bin/bash
# Automated backup with S3 upload and local rotation

set -euo pipefail

# ─── Configuration ───
BENCH_PATH="$HOME/frappe-bench"
SITES=("scoopjoy.com" "outlet1.scoopjoy.com" "outlet2.scoopjoy.com")
S3_BUCKET="s3://scoopjoy-erp-backups"
S3_REGION="us-east-1"
LOCAL_RETENTION_DAYS=7
S3_RETENTION_DAYS=90
LOG_FILE="${BENCH_PATH}/logs/s3-backup.log"
DATE=$(date +%Y-%m-%d_%H-%M-%S)

log() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$LOG_FILE"
}

backup_site() {
    local site=$1
    log "Starting backup for ${site}"

    cd "$BENCH_PATH"
    ./env/bin/bench --site "$site" backup --with-files 2>> "$LOG_FILE"

    if [ $? -ne 0 ]; then
        log "ERROR: Backup failed for ${site}"
        return 1
    fi

    local backup_dir="${BENCH_PATH}/sites/${site}/private/backups"
    local latest_db=$(ls -t "${backup_dir}"/*-database.sql.gz 2>/dev/null | head -1)
    local latest_files=$(ls -t "${backup_dir}"/*-files.tar 2>/dev/null | head -1)
    local latest_private=$(ls -t "${backup_dir}"/*-private-files.tar 2>/dev/null | head -1)

    local s3_path="${S3_BUCKET}/${site}/${DATE}"

    [ -n "$latest_db" ] && aws s3 cp "$latest_db" "${s3_path}/" --region "$S3_REGION" 2>> "$LOG_FILE" && log "Uploaded database backup: $(basename "$latest_db")"
    [ -n "$latest_files" ] && aws s3 cp "$latest_files" "${s3_path}/" --region "$S3_REGION" 2>> "$LOG_FILE" && log "Uploaded files backup: $(basename "$latest_files")"
    [ -n "$latest_private" ] && aws s3 cp "$latest_private" "${s3_path}/" --region "$S3_REGION" 2>> "$LOG_FILE" && log "Uploaded private files backup: $(basename "$latest_private")"

    # site_config.json holds the encryption key
    aws s3 cp "${BENCH_PATH}/sites/${site}/site_config.json" \
        "${s3_path}/site_config.json" --region "$S3_REGION" 2>> "$LOG_FILE"

    log "Backup complete for ${site}"
}

cleanup_local() {
    log "Cleaning local backups older than ${LOCAL_RETENTION_DAYS} days"
    for site in "${SITES[@]}"; do
        find "${BENCH_PATH}/sites/${site}/private/backups" \
            -type f -mtime "+${LOCAL_RETENTION_DAYS}" -delete 2>> "$LOG_FILE"
    done
}

cleanup_s3() {
    log "Cleaning S3 backups older than ${S3_RETENTION_DAYS} days"
    local cutoff_date=$(date -d "-${S3_RETENTION_DAYS} days" +%Y-%m-%d)
    for site in "${SITES[@]}"; do
        aws s3 ls "${S3_BUCKET}/${site}/" --region "$S3_REGION" | \
            while read -r line; do
                local dir_date=$(echo "$line" | awk '{print $2}' | cut -d'_' -f1 | tr -d '/')
                if [[ "$dir_date" < "$cutoff_date" ]]; then
                    local dir_name=$(echo "$line" | awk '{print $2}')
                    aws s3 rm "${S3_BUCKET}/${site}/${dir_name}" \
                        --recursive --region "$S3_REGION" 2>> "$LOG_FILE"
                    log "Removed old S3 backup: ${site}/${dir_name}"
                fi
            done
    done
}

log "========================================="
log "Starting backup run"

for site in "${SITES[@]}"; do
    backup_site "$site"
done

cleanup_local
cleanup_s3

log "Backup run complete"
log "========================================="
```

```bash
chmod +x ~/scripts/backup-to-s3.sh
crontab -e
# 0 */6 * * * ~/scripts/backup-to-s3.sh
```

## Restoring from backup

```bash
# Full restore: DB + public + private files
bench --site scoopjoy.com --force restore \
  /path/to/20260320_120000-scoopjoy-database.sql.gz \
  --with-public-files /path/to/20260320_120000-scoopjoy-files.tar \
  --with-private-files /path/to/20260320_120000-scoopjoy-private-files.tar

# After restore
bench --site scoopjoy.com migrate
bench --site scoopjoy.com clear-cache
bench build
bench restart
```

:::danger[Restore the encryption key, or lose your secrets]
Copy the original site's `encryption_key` from the backed-up `site_config.json` into the new site's config. Without it, encrypted fields (passwords, API keys) fail to decrypt — permanently inaccessible even though the rows are intact.
:::

### Restoring in Docker

Same flow via `docker compose exec` into the `backend` container:

```bash
aws s3 cp s3://scoopjoy-erp-backups/scoopjoy.com/2026-03-20_02-00-00/ ./restore/ --recursive

docker compose exec backend bench --site scoopjoy.com --force restore \
  /home/frappe/frappe-bench/sites/restore/20260320-database.sql.gz \
  --with-public-files /home/frappe/frappe-bench/sites/restore/20260320-files.tar \
  --with-private-files /home/frappe/frappe-bench/sites/restore/20260320-private-files.tar

docker compose exec backend bench --site scoopjoy.com migrate
docker compose exec backend bench --site scoopjoy.com clear-cache
```

## Disaster recovery

| Metric | Target | Implementation |
|--------|--------|----------------|
| **RPO** | 6 hours | Backup every 6 hours to S3 |
| **RTO** | 2 hours | Documented runbook + tested restore |

### MariaDB replication (near-zero RPO)

Primary:

```ini title="/etc/mysql/mariadb.conf.d/99-replication.cnf (primary)"
[mysqld]
server-id              = 1
log_bin                = /var/log/mysql/mariadb-bin
binlog_format          = ROW
expire_logs_days       = 7
max_binlog_size        = 100M
binlog_do_db           = _scoopjoy_com
binlog_do_db           = _outlet1_scoopjoy_com
```

```sql
-- On primary: replication user + position
CREATE USER 'replication'@'%' IDENTIFIED BY 'secure-repl-password';
GRANT REPLICATION SLAVE ON *.* TO 'replication'@'%';
FLUSH PRIVILEGES;
SHOW MASTER STATUS;
```

Replica:

```ini title="/etc/mysql/mariadb.conf.d/99-replication.cnf (replica)"
[mysqld]
server-id              = 2
relay_log              = /var/log/mysql/relay-bin
read_only              = 1
```

```sql
-- On replica
CHANGE MASTER TO
  MASTER_HOST='primary-db.example.com',
  MASTER_USER='replication',
  MASTER_PASSWORD='secure-repl-password',
  MASTER_LOG_FILE='mariadb-bin.000001',
  MASTER_LOG_POS=XXX;
START SLAVE;
SHOW SLAVE STATUS\G
```

### Disaster-recovery runbook

Keep this tested — an untested backup is just a hope:

```text title="DISASTER RECOVERY RUNBOOK — ScoopJoy ERP"
SCENARIO 1: Application server failure (OS/hardware)
─────────────────────────────────────────────────────
1. Provision new Ubuntu 24.04 server
2. Run production-setup.sh (Chapter 27)
3. aws s3 cp s3://scoopjoy-erp-backups/scoopjoy.com/LATEST/ ./restore/ --recursive
4. Restore each site:
   bench --site scoopjoy.com --force restore ./restore/*database.sql.gz \
     --with-public-files ./restore/*files.tar \
     --with-private-files ./restore/*private-files.tar
5. Copy encryption_key from backed-up site_config.json
6. bench --site scoopjoy.com migrate
7. bench restart
8. Update DNS to new server IP
9. sudo -H bench setup lets-encrypt scoopjoy.com
ESTIMATED TIME: 1-2 hours

SCENARIO 2: Database corruption
──────────────────────────────
1. sudo supervisorctl stop all
2. Download latest S3 database backup
3. bench --site scoopjoy.com --force restore ./restore/*database.sql.gz
4. bench --site scoopjoy.com migrate
5. bench --site scoopjoy.com clear-cache
6. sudo supervisorctl start all
7. Spot-check data integrity
ESTIMATED TIME: 30-60 minutes

SCENARIO 3: Accidental data deletion by user
────────────────────────────────────────────
1. Find the most recent backup BEFORE the deletion
2. bench new-site recovery.localhost --mariadb-root-password XXX
3. Restore the backup to the recovery site
4. Use bench console to extract the deleted records
5. Re-create the records on production
6. bench drop-site recovery.localhost --force
ESTIMATED TIME: 1-3 hours depending on data volume

SCENARIO 4: Complete infrastructure failure
──────────────────────────────────────────
1. Follow Scenario 1 on a new provider
2. If using K8s: redeploy the Helm chart to a new cluster (Chapter 28)
3. Restore database + files from S3
4. Update DNS records (keep TTL low: 300s)
5. Verify all sites and services
ESTIMATED TIME: 2-4 hours
```

### Prometheus + Grafana

Scrape system metrics (node_exporter), MariaDB, Redis, Nginx, and Frappe's ping endpoint:

```yaml title="/etc/prometheus/prometheus.yml"
global:
  scrape_interval: 15s

scrape_configs:
  - job_name: 'node'      # CPU, memory, disk
    static_configs:
      - targets: ['localhost:9100']
    relabel_configs:
      - source_labels: [__address__]
        target_label: instance
        replacement: 'scoopjoy-erp-server'

  - job_name: 'mariadb'
    static_configs:
      - targets: ['localhost:9104']

  - job_name: 'redis'
    static_configs:
      - targets: ['localhost:9121']

  - job_name: 'nginx'
    static_configs:
      - targets: ['localhost:9113']

  - job_name: 'frappe'
    metrics_path: /api/method/ping
    static_configs:
      - targets: ['localhost:8000']
```

```bash
# Node exporter (system metrics)
sudo apt install prometheus-node-exporter

# MariaDB exporter — create a monitoring user first:
#   CREATE USER 'exporter'@'localhost' IDENTIFIED BY 'password';
#   GRANT PROCESS, REPLICATION CLIENT, SELECT ON *.* TO 'exporter'@'localhost';
docker run -d --name mariadb-exporter \
  -p 9104:9104 \
  -e DATA_SOURCE_NAME="exporter:password@(localhost:3306)/" \
  prom/mysqld-exporter

# Redis exporter
docker run -d --name redis-exporter \
  -p 9121:9121 \
  oliver006/redis_exporter \
  --redis.addr=redis://localhost:6379
```

Queue depth in Grafana, as PromQL:

```text title="PromQL — pending RQ jobs"
redis_key_size{key=~"rq:queue:.*"}
```

### Key metrics to alert on

| Metric | Source | Alert threshold |
|--------|--------|-----------------|
| Response time (P95) | Nginx access log | > 5 s |
| Worker queue depth | Redis `rq:queue:default` | > 100 pending |
| DB size | MariaDB `information_schema` | > 80% disk |
| CPU usage | node_exporter | > 85% sustained |
| Memory usage | node_exporter | > 90% |
| Disk I/O wait | node_exporter | > 20% |
| SSL cert expiry | blackbox_exporter | < 14 days |
| Failed logins | Frappe Activity Log | > 50/hour |
| Background job failures | worker logs | > 10/hour |

Centralize with Filebeat → Elasticsearch/OpenSearch:

```yaml title="/etc/filebeat/filebeat.yml"
filebeat.inputs:
  - type: log
    enabled: true
    paths:
      - /home/frappe/frappe-bench/logs/*.log
    fields:
      service: frappe-bench
    multiline:
      pattern: '^\['
      negate: true
      match: after

  - type: log
    enabled: true
    paths:
      - /home/frappe/frappe-bench/sites/*/logs/frappe.log
    fields:
      service: frappe-app
    multiline:
      pattern: '^Traceback'
      negate: true
      match: after

output.elasticsearch:
  hosts: ["https://elasticsearch.example.com:9200"]
  index: "frappe-logs-%{+yyyy.MM.dd}"
```

### Cron health-check script

No Prometheus? A cron script covers the essentials — web ping, Supervisor, queue depth, disk, MariaDB, Redis, SSL expiry — and alerts to Slack:

```bash title="~/scripts/health-check.sh"
#!/bin/bash
# Lightweight health check with Slack alerting

set -euo pipefail

SITE="scoopjoy.com"
BENCH_PATH="$HOME/frappe-bench"
SLACK_WEBHOOK="${SLACK_WEBHOOK_URL:-}"

alert() {
    local level=$1
    local message=$2
    echo "[$(date)] ${level}: ${message}"

    if [ -n "$SLACK_WEBHOOK" ]; then
        local color="danger"
        [ "$level" = "WARNING" ] && color="warning"
        [ "$level" = "OK" ] && color="good"

        curl -s -X POST "$SLACK_WEBHOOK" \
            -H 'Content-type: application/json' \
            -d "{\"attachments\":[{\"color\":\"${color}\",\"title\":\"ERPNext Health: ${level}\",\"text\":\"${message}\",\"footer\":\"${SITE}\"}]}" \
            > /dev/null 2>&1
    fi
}

# 1. Web server responding
HTTP_CODE=$(curl -s -o /dev/null -w '%{http_code}' "http://localhost:8000/api/method/ping" || echo "000")
[ "$HTTP_CODE" != "200" ] && alert "CRITICAL" "Web server not responding (HTTP ${HTTP_CODE})"

# 2. Supervisor processes
STOPPED=$(sudo supervisorctl status | grep -c "STOPPED\|FATAL\|EXITED" || true)
if [ "$STOPPED" -gt 0 ]; then
    STOPPED_NAMES=$(sudo supervisorctl status | grep "STOPPED\|FATAL\|EXITED" | awk '{print $1}')
    alert "CRITICAL" "Supervisor processes down: ${STOPPED_NAMES}"
fi

# 3. Worker queue depth
QUEUE_DEPTH=$(cd "$BENCH_PATH" && ./env/bin/python -c "
import redis
r = redis.Redis()
print(r.llen('rq:queue:default') + r.llen('rq:queue:short') + r.llen('rq:queue:long'))
" 2>/dev/null || echo "0")
[ "$QUEUE_DEPTH" -gt 100 ] && alert "WARNING" "Worker queue depth high: ${QUEUE_DEPTH} jobs"

# 4. Disk usage
DISK_USAGE=$(df -h / | awk 'NR==2 {print $5}' | tr -d '%')
[ "$DISK_USAGE" -gt 85 ] && alert "WARNING" "Disk usage at ${DISK_USAGE}%"

# 5. MariaDB / 6. Redis
systemctl is-active --quiet mariadb || alert "CRITICAL" "MariaDB is not running"
systemctl is-active --quiet redis-server || alert "CRITICAL" "Redis is not running"

# 7. SSL certificate expiry
CERT_EXPIRY=$(echo | openssl s_client -servername "$SITE" -connect "$SITE":443 2>/dev/null | \
    openssl x509 -noout -enddate 2>/dev/null | cut -d= -f2)
if [ -n "$CERT_EXPIRY" ]; then
    DAYS_LEFT=$(( ($(date -d "$CERT_EXPIRY" +%s) - $(date +%s)) / 86400 ))
    [ "$DAYS_LEFT" -lt 14 ] && alert "WARNING" "SSL certificate expires in ${DAYS_LEFT} days"
fi
```

```bash
chmod +x ~/scripts/health-check.sh
crontab -e
# */5 * * * * SLACK_WEBHOOK_URL="https://hooks.slack.com/services/XXX/YYY/ZZZ" ~/scripts/health-check.sh >> ~/frappe-bench/logs/health-check.log 2>&1
```
