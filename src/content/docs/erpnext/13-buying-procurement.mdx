## The purchase cycle

<Mermaid title="Material Request / RFQ → Supplier Quotation → Purchase Order → Purchase Receipt (+ Quality Inspection) → Purchase Invoice → Payment Entry" height={420} code={`
flowchart LR
  MR["Material Request"] --> SQ["Supplier Quotation"]
  MR -.-> RFQ["Request for Quotation"]
  RFQ --> SQ
  SQ --> PO["Purchase Order"]

  PR --> QI["Quality Inspection"]
  PR --> PI["Purchase Invoice"]
  PI --> PE["Payment Entry"]

| Document | Purpose | Accounting impact |
|---|---|---|
| Material Request | Internal request for materials | None |
| Request for Quotation | Sent to suppliers for quotes | None |
| Supplier Quotation | Supplier's price response | None |
| Purchase Order | Confirmed order to supplier | None (commitment) |
| Purchase Receipt | Goods received at warehouse | Stock increased |
| Purchase Invoice | Supplier's bill recorded | Expense/asset booked, payable created |
| Payment Entry | Payment made to supplier | Payable settled |

Early stages are commitments, not ledger entries — accounting starts at receipt/invoice.

def create_ingredient_supplier(name, group, lead_time_days=7):

## Request for Quotation (RFQ)

Send one RFQ to multiple suppliers, then `submit()` + `send_to_supplier()` to email them. Suppliers respond via the **Supplier Portal** (if enabled) or manually as Supplier Quotations:

```python title="Sending an RFQ to three packaging suppliers"
def create_rfq_for_packaging():
    """Send RFQ to 3 suppliers for packaging materials."""
    rfq = frappe.new_doc("Request for Quotation")
    rfq.company = "ScoopJoy Ice Creams Pvt Ltd"
    rfq.transaction_date = frappe.utils.today()
    rfq.message_for_supplier = (
        "Please provide your best quote for the following "
        "packaging materials. Delivery required within 14 days."
    )

    rfq.append("items", {
        "item_code": "PKG-TUB-500",
        "item_name": "Ice Cream Tub 500ml",
        "qty": 10000,
        "warehouse": "Central Warehouse - SJ",
        "uom": "Nos",
        "schedule_date": frappe.utils.add_days(frappe.utils.today(), 14)
    })
    rfq.append("items", {
        "item_code": "PKG-LID-500",
        "item_name": "Tub Lid 500ml",
        "qty": 10000,
        "warehouse": "Central Warehouse - SJ",
        "uom": "Nos",
        "schedule_date": frappe.utils.add_days(frappe.utils.today(), 14)
    })

    rfq.append("suppliers", {"supplier": "SUP-PKG-01"})  # PackWell India
    rfq.append("suppliers", {"supplier": "SUP-PKG-02"})  # GreenPack Ltd
    rfq.append("suppliers", {"supplier": "SUP-PKG-03"})  # QuickPack Co

    rfq.insert()
    rfq.submit()
    rfq.send_to_supplier()  # email suppliers

    return rfq.name
```

Compare responses side-by-side at **Buying › Supplier Quotation › Compare Supplier Quotations**.

## Purchase Order + approval workflow

```python title="Creating a Purchase Order"
def create_purchase_order(supplier, items, delivery_date):
    """Create a PO from the selected supplier quotation."""
    po = frappe.new_doc("Purchase Order")
    po.supplier = supplier
    po.company = "ScoopJoy Ice Creams Pvt Ltd"
    po.schedule_date = delivery_date

    for item in items:
        po.append("items", {
            "item_code": item["item_code"],
            "qty": item["qty"],
            "rate": item["rate"],
            "warehouse": "Central Warehouse - SJ",
            "schedule_date": delivery_date
        })

    po.insert()
    return po.name
```

### Workflow for high-value POs

A Workflow defines **states** (each maps to a `doc_status` and an `allow_edit` role) and **transitions** (actions gated by role + optional condition). POs above ₹50,000 need manager approval:

```python title="Purchase Order approval workflow"
def setup_po_approval_workflow():
    """POs above INR 50,000 require Purchase Manager approval."""
    workflow = frappe.new_doc("Workflow")
    workflow.workflow_name = "Purchase Order Approval"
    workflow.document_type = "Purchase Order"
    workflow.is_active = 1
    workflow.send_email_alert = 1

    # States
    workflow.append("states", {"state": "Draft", "doc_status": "0", "allow_edit": "Purchase User"})
    workflow.append("states", {"state": "Pending Approval", "doc_status": "0", "allow_edit": "Purchase Manager"})
    workflow.append("states", {"state": "Approved", "doc_status": "1", "allow_edit": "Purchase Manager"})
    workflow.append("states", {"state": "Rejected", "doc_status": "0", "allow_edit": "Purchase Manager"})  # back to draft

    # Transitions
    workflow.append("transitions", {
        "state": "Draft", "action": "Submit for Approval",
        "next_state": "Pending Approval",
        "allowed": "Purchase User",
        "condition": "doc.grand_total > 50000"
    })
    workflow.append("transitions", {
        "state": "Draft", "action": "Submit",
        "next_state": "Approved",
        "allowed": "Purchase User",
        "condition": "doc.grand_total <= 50000"
    })
    workflow.append("transitions", {
        "state": "Pending Approval", "action": "Approve",
        "next_state": "Approved", "allowed": "Purchase Manager"
    })
    workflow.append("transitions", {
        "state": "Pending Approval", "action": "Reject",
        "next_state": "Rejected", "allowed": "Purchase Manager"
    })

    workflow.insert()
    return workflow.name
```

Pair it with **Buying › Authorization Rule** as a safety net — it can block POs above an amount unless approved by a specific role, even without a workflow.

## Purchase Receipt & quality inspection

`make_purchase_receipt` pre-fills items from the PO — adjust received quantities, then submit:

```python title="Receiving goods against a PO"
from erpnext.buying.doctype.purchase_order.purchase_order import make_purchase_receipt

pr = make_purchase_receipt("PO-00042")
for item in pr.items:
    item.received_qty = item.qty  # full delivery
    item.qty = item.qty
pr.insert()
pr.submit()
```

For inspected items (e.g. milk), enable inspection on the Item master — the Purchase Receipt then can't submit until a Quality Inspection passes:

## Subcontracting

Send raw materials to a supplier who processes them into finished goods. v15/v16 use the **Subcontracting Order** DocType (the legacy PO-based `is_subcontracted` approach is deprecated).

Mark the item for subcontracting with the v14+ field `supply_raw_materials_for_purchase` (the legacy `is_sub_contracted_item` flag is gone) and set a default BOM:

```python title="Configuring a subcontracted item"
item = frappe.get_doc("Item", "RM-MANGO-PULP-PASTEURIZED")
item.supply_raw_materials_for_purchase = 1
item.default_bom = "BOM-RM-MANGO-PULP-PASTEURIZED-001"
item.save()
# BOM: Pasteurized Mango Pulp ← Raw Mango Pulp 1.1 kg (10% processing loss)
```

## Supplier scorecard

Weighted criteria produce a single grade; **standings** map grade bands to actions — a poor grade blocks future RFQs and POs:

```python title="Setting up a supplier scorecard"
def setup_supplier_scorecard(supplier):
    scorecard = frappe.new_doc("Supplier Scorecard")
    scorecard.supplier = supplier
    scorecard.weighting_function = "{total_score} / {max_score} * 100"

    scorecard.append("criteria", {"criteria_name": "Delivery Timeliness", "weight": 40})
    scorecard.append("criteria", {"criteria_name": "Quality of Materials", "weight": 35})
    scorecard.append("criteria", {"criteria_name": "Price Competitiveness", "weight": 25})

    scorecard.append("standings", {"standing_name": "Excellent", "standing_color": "Green", "min_grade": 80, "max_grade": 100})
    scorecard.append("standings", {"standing_name": "Acceptable", "standing_color": "Yellow", "min_grade": 50, "max_grade": 80})
    scorecard.append("standings", {
        "standing_name": "Poor", "standing_color": "Red",
        "min_grade": 0, "max_grade": 50,
        "prevent_rfqs": 1, "prevent_pos": 1
    })

    scorecard.insert()
    return scorecard.name
```

Criteria formulas use built-in variables like `{total_accepted_items}` and `{total_received_items}`:

```text title="Delivery timeliness formula"
({total_accepted_items} / {total_received_items}) * 100
  if {total_received_items} > 0 else 100
```

## Buying Settings

| Setting | Recommended value | Purpose |
|---|---|---|
| Supplier Naming By | Naming Series | Auto-generate IDs |
| Default Supplier Group | All Supplier Groups | Catch-all |
| Buying Price List | Standard Buying | Default price list |
| Purchase Order Required | Yes | PO before receipt |
| Purchase Receipt Required | Yes | Receipt before invoice |
| Maintain Same Rate Throughout Purchase Cycle | Yes | Prevent rate changes |

## Practical examples

### Ingredient procurement with reorder levels

```python title="Setting up ingredient procurement"
def setup_ingredient_procurement():
    """Set up core ingredient items with reorder levels for auto-replenishment."""
    ingredients = [
        {"item_code": "RM-MILK-WHOLE", "item_name": "Whole Milk", "item_group": "Raw Materials - Dairy",
         "stock_uom": "Litre", "safety_stock": 500, "reorder_level": 200, "reorder_qty": 1000, "lead_time_days": 1},
        {"item_code": "RM-SUGAR-WHITE", "item_name": "White Sugar", "item_group": "Raw Materials - Sweeteners",
         "stock_uom": "Kg", "safety_stock": 200, "reorder_level": 100, "reorder_qty": 500, "lead_time_days": 3},
        {"item_code": "RM-VANILLA-EXTRACT", "item_name": "Vanilla Extract", "item_group": "Raw Materials - Flavoring",
         "stock_uom": "Litre", "safety_stock": 20, "reorder_level": 10, "reorder_qty": 50, "lead_time_days": 7}
    ]

    for ing in ingredients:
        if not frappe.db.exists("Item", ing["item_code"]):
            item = frappe.new_doc("Item")
            item.item_code = ing["item_code"]
            item.item_name = ing["item_name"]
            item.item_group = ing["item_group"]
            item.stock_uom = ing["stock_uom"]
            item.is_stock_item = 1

            item.append("reorder_levels", {
                "warehouse": "Central Warehouse - SJ",
                "warehouse_reorder_level": ing["reorder_level"],
                "warehouse_reorder_qty": ing["reorder_qty"],
                "material_request_type": "Purchase"
            })

            item.safety_stock = ing["safety_stock"]
            item.lead_time_days = ing["lead_time_days"]
            item.insert()

    frappe.db.commit()
```

`safety_stock` covers demand variability — the reorder calc is: reorder if `projected_qty < reorder_level + safety_stock`.

### RFQ → quotes → compare → PO

<Steps>

1. **Create the RFQ** (code above).

2. **Enter supplier responses** via `make_supplier_quotation_from_rfq`:

   ```python title="Entering supplier responses to an RFQ"
   def enter_supplier_quotation(rfq_name, supplier, quotes):
       from erpnext.buying.doctype.request_for_quotation.request_for_quotation import (
           make_supplier_quotation_from_rfq,
       )
       sq = make_supplier_quotation_from_rfq(rfq_name, for_supplier=supplier)
       for i, item in enumerate(sq.items):
           item.rate = quotes[i]["rate"]
           item.lead_time_days = quotes[i].get("lead_time_days", 14)
       sq.insert()
       sq.submit()
       return sq.name

   enter_supplier_quotation("RFQ-00001", "SUP-PKG-01", [
       {"rate": 3.50, "lead_time_days": 10}, {"rate": 1.20, "lead_time_days": 10}])
   enter_supplier_quotation("RFQ-00001", "SUP-PKG-02", [
       {"rate": 3.20, "lead_time_days": 14}, {"rate": 1.35, "lead_time_days": 14}])  # cheapest tubs
   enter_supplier_quotation("RFQ-00001", "SUP-PKG-03", [
       {"rate": 3.80, "lead_time_days": 7}, {"rate": 1.10, "lead_time_days": 7}])     # fastest
   ```

3. **Compare and select** at **Buying › Supplier Quotation › Compare Supplier Quotations**.

4. **Create the PO** from the chosen quotation:

   ```python title="Creating a PO from the chosen quotation"

   po.insert()
   po.submit()
   ```

</Steps>

### Auto Material Request from reorder levels

Enable in **Stock Settings**: `auto_indent = 1` (Auto Material Request) and `reorder_email_notify = 1`. The scheduler runs `reorder_item()` periodically — when projected qty falls below the reorder level, a Material Request (Purchase/Manufacture/Transfer per item config) is auto-created:

```python title="Enabling auto Material Requests + checking projected qty"
stock_settings = frappe.get_single("Stock Settings")
stock_settings.auto_indent = 1
stock_settings.reorder_email_notify = 1
stock_settings.save()

from erpnext.stock.utils import get_latest_stock_qty
projected = frappe.db.get_value("Bin", {
    "item_code": "RM-MILK-WHOLE",
    "warehouse": "Central Warehouse - SJ"
}, "projected_qty")
print(f"Projected qty: {projected}")  # if < reorder_level, a Material Request is auto-created
```
